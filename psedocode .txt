DECLARE n : INTEGER
DECLARE i : INTEGER
DECLARE usia : INTEGER
DECLARE harga : REAL
DECLARE total : REAL
DECLARE diskon : REAL

total <- 0

OUTPUT "Masukkan jumlah pengunjung:"
INPUT n

FOR i <- 1 TO n
    OUTPUT "Masukkan usia pengunjung ke-"
    OUTPUT i
    INPUT usia

    IF usia < 5 THEN
        harga <- 0
    ELSE
        IF usia <= 12 THEN
            harga <- 5
        ELSE
            IF usia <= 59 THEN
                harga <- 10
            ELSE
                harga <- 7
            ENDIF
        ENDIF
    ENDIF

    total <- total + harga
NEXT i

IF n >= 5 THEN
    diskon <- total * 0.10
    total <- total - diskon
ENDIF

OUTPUT "Jumlah tiket:"
OUTPUT n
OUTPUT "Total harga akhir:"
OUTPUT total
