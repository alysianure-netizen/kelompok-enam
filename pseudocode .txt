START

DECLARE jumlahPengunjung, usia, hargaTiket, totalHarga, diskon, i
SET totalHarga = 0

// 1. Menanyakan jumlah pengunjung
INPUT jumlahPengunjung

// 2 & 3. Menentukan harga tiket tiap pengunjung dan menghitung total
FOR i = 1 TO jumlahPengunjung DO
    INPUT usia (untuk pengunjung ke-i)
    
    Tentukan hargaTiket berdasarkan usia:
        - usia di bawah 5 tahun -> hargaTiket = 0 (gratis)
        - usia 5 sampai 12 tahun -> hargaTiket = 5
        - usia 13 sampai 59 tahun -> hargaTiket = 10
        - usia 60 tahun atau lebih -> hargaTiket = 7
    
    totalHarga = totalHarga + hargaTiket
END FOR

// 4. Memberikan diskon jika memenuhi ketentuan
Tentukan diskon:
    - jika jumlahPengunjung 5 orang atau lebih -> diskon = totalHarga * 10 / 100, lalu totalHarga = totalHarga - diskon
    - jika kurang dari 5 -> diskon = 0

// 5. Menampilkan jumlah tiket dan harga akhir
OUTPUT "Jumlah tiket: " + jumlahPengunjung
OUTPUT "Diskon: $" + diskon
OUTPUT "Total harga akhir: $" + totalHarga

END
